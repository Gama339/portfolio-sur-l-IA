Génération de codes aléatoires et insertion dans MySQL


                        # Importation des bibliothèques nécessaires
                        import random
                        import mysql.connector


                        # ==================================================
                        # 1. CONNEXION À LA BASE DE DONNÉES MYSQL
                        # ==================================================

                        connexion = mysql.connector.connect(
                            host="localhost",       # Adresse IP du serveur MySQL
                            user="mon_utilisateur",                # Nom d'utilisateur MySQL
                            password="mon_mot_de_passe",            # Mot de passe MySQL
                            database="mon_base_de_donnees"              # Nom de la base de données
                        )

                        # Création d'un curseur pour exécuter les requêtes SQL
                        curseur = connexion.cursor()


                        try:

                            # ==================================================
                            # 2. PRÉPARATION DE LA REQUÊTE SQL
                            # ==================================================

                            # INSERT IGNORE permet d'ignorer les doublons
                            # si la colonne resultat possède une contrainte UNIQUE.
                            requete = """
                            INSERT IGNORE INTO resultats (resultat)
                                VALUES (%s)
                            """

                            # Compteur du nombre de tentatives effectuées
                            tentatives = 0

                            # Compteur du nombre de lignes réellement insérées
                            insertions = 0


                            # ==================================================
                            # 3. GÉNÉRATION DES RÉSULTATS ALÉATOIRES
                            # ==================================================

                            # La boucle effectue 10 000 tentatives
                            for i in range(10000):

                                # Génération de quatre chiffres entre 0 et 9
                                A = random.randint(0, 9)
                                B = random.randint(0, 9)
                                C = random.randint(0, 9)
                                D = random.randint(0, 9)

                                # Concaténation des quatre chiffres
                                # Exemple : 3, 7, 0, 5 devient "3705"
                                resultat = str(A) + str(B) + str(C) + str(D)

                                # Affichage du résultat généré
                                print(resultat)

                                # ==================================================
                                # 4. INSERTION DANS MYSQL
                                # ==================================================

                                # Le résultat est transmis sous forme de tuple.
                                # La virgule est nécessaire pour un tuple à un élément.
                                valeurs = (resultat,)

                                # Exécution de la requête SQL
                                curseur.execute(requete, valeurs)

                                # Comptabilisation de la tentative
                                tentatives += 1

                                # rowcount vaut généralement 1 si une ligne a été insérée,
                                # et 0 si INSERT IGNORE a ignoré un doublon.
                                insertions += curseur.rowcount


                        # ==================================================
                        # 5. VALIDATION DES INSERTIONS
                        # ==================================================

                        # Enregistrement définitif des modifications dans MySQL
                        connexion.commit()

                        # Affichage du bilan
                        print("\n========== BILAN ==========")
                        print("Tentatives effectuées :", tentatives)
                        print("Nouvelles lignes insérées :", insertions)
                        print("Doublons ignorés :", tentatives - insertions)


                        # ==================================================
                        # 6. GESTION DES ERREURS MYSQL
                        # ==================================================

                        except mysql.connector.Error as erreur:

                            # Annulation des modifications non validées
                            connexion.rollback()

                            # Affichage du message d'erreur
                            print("Erreur MySQL :", erreur)


                        # ==================================================
                        # 7. FERMETURE DES RESSOURCES
                        # ==================================================

                        finally:

                        # Fermeture du curseur
                        curseur.close()

                        # Fermeture de la connexion MySQL
                        connexion.close()

                        print("Connexion MySQL fermée.")
